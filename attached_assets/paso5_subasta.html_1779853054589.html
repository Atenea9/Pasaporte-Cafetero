export const ScaFormScreen = () => {
  const navigation = useNavigation();
  const { user } = useAuth();
  
  const [scores, setScores] = useState({
    aroma: '0', flavor: '0', aftertaste: '0', acidity: '0', body: '0',
    balance: '0', uniformity: '10', cleanCup: '10', sweetness: '10', overall: '0'
  });

  const calculateTotal = () => {
    return Object.values(scores).reduce((acc, curr) => acc + (parseFloat(curr) || 0), 0).toFixed(2);
  };

    await mockDbService.submitScaAnalysis(user!.uid, scores);
    Alert.alert('Enviado', 'Análisis SCA enviado para revisión.', [{ text: 'OK', onPress: () => navigation.goBack() }]);
  };

  const renderInput = (label: string, key: keyof typeof scores) => (
    
      {label}
       setScores({...scores, [key]: val})}

  );
