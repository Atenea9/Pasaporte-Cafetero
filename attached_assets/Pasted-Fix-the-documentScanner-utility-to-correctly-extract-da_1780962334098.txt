// NUIP / número de documento
// Real OCR output example: "NUIP 1.999.999.999"
const nuipMatch = rawText.match(/NUIP\s+([\d\.]+)/i);
if (nuipMatch) numero_documento = nuipMatch[1].replace(/\./g, '');

// Apellidos — line after "Apellidos" label OR all-caps line near top-right
// Real OCR output example: "VELEZ RUIZ"
const apellidosMatch = rawText.match(/Apellidos?\s*\n?\s*([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ\s]+)/i);
if (apellidosMatch) apellidos = apellidosMatch[1].trim();

// Nombres — line after "Nombres" label
// Real OCR output: label is "Nombres" then next line has the name
const nombresMatch = rawText.match(/Nombres?\s*\n?\s*([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ\s]+)/i);
if (nombresMatch) nombres = nombresMatch[1].trim();

// Fecha de nacimiento
// Real OCR output: "12MAR2000" or "12 MAR 2000" or "12MAR 2000"
const fechaMatch = rawText.match(/(\d{1,2})\s*(ENE|FEB|MAR|ABR|MAY|JUN|JUL|AGO|SEP|OCT|NOV|DIC)\s*(\d{4})/i);
if (fechaMatch) {
  const months = {ENE:'01',FEB:'02',MAR:'03',ABR:'04',MAY:'05',JUN:'06',
                  JUL:'07',AGO:'08',SEP:'09',OCT:'10',NOV:'11',DIC:'12'};
  fecha_nacimiento = `${fechaMatch[3]}-${months[fechaMatch[2].toUpperCase()]}-${fechaMatch[1].padStart(2,'0')}`;
}

// Lugar de nacimiento: "BOGOTA D.C. (CUNDINAMARCA)"
// Extract municipio (before parenthesis) and departamento (inside parenthesis)
const lugarMatch = rawText.match(/([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ\s\.]+)\s*\(([A-ZÁÉÍÓÚÑ\s]+)\)/);
if (lugarMatch) {
  municipio_ciudad = lugarMatch[1].trim();
  region_departamento = lugarMatch[2].trim();
}

// País: always Colombia if REPUBLICA DE COLOMBIA found
if (/COLOMBIA/i.test(rawText)) pais_emision = 'Colombia';

const wordsInZone = (yMin, yMax) => data.words

if (!apellidos) apellidos = wordsInZone(120, 160);
